# 11.扩展插件

### 一、分页插件（PaginationInnerInterceptor）
```java
@Configuration
public class MybatisPlusConfig {
    @Bean
    public MybatisPlusInterceptor mybatisPlusInterceptor() {
        MybatisPlusInterceptor interceptor = new MybatisPlusInterceptor();
        // 分页插件（需设置数据库类型）
        interceptor.addInnerInterceptor(new PaginationInnerInterceptor(DbType.MYSQL));
        return interceptor;
    }
}

// 使用示例
Page<User> page = new Page<>(1, 10); // 当前页/每页条数
userMapper.selectPage(page, Wrappers.<User>query().eq("status", 1));
```

### 二、乐观锁插件（OptimisticLockerInnerInterceptor）
```java
// 实体类添加版本字段
@Data
public class User {
    @Version
    private Integer version;
}

// 配置类
@Bean
public MybatisPlusInterceptor mybatisPlusInterceptor() {
    MybatisPlusInterceptor interceptor = new MybatisPlusInterceptor();
    interceptor.addInnerInterceptor(new OptimisticLockerInnerInterceptor());
    return interceptor;
}

// 测试并发场景
User user1 = userMapper.selectById(1L); // version=1
User user2 = userMapper.selectById(1L); // version=1

user1.setName("A");
userMapper.updateById(user1); // 成功 version=2

user2.setName("B");
userMapper.updateById(user2); // 失败！version已变更
```

### 三、防全表操作插件（BlockAttackInnerInterceptor）
```java
@Bean
public MybatisPlusInterceptor mybatisPlusInterceptor() {
    MybatisPlusInterceptor interceptor = new MybatisPlusInterceptor();
    interceptor.addInnerInterceptor(new BlockAttackInnerInterceptor());
    return interceptor;
}

// 测试非法操作（会抛出异常）
userMapper.delete(null); // 抛出MybatisPlusException
userMapper.update(new User(), null); // 抛出异常
```

### 四、动态表名插件（DynamicTableNameInnerInterceptor）
```java
@Bean
public MybatisPlusInterceptor mybatisPlusInterceptor() {
    MybatisPlusInterceptor interceptor = new MybatisPlusInterceptor();
    DynamicTableNameInnerInterceptor dynamicInterceptor = new DynamicTableNameInnerInterceptor();
    
    dynamicInterceptor.setTableNameHandler((sql, tableName) -> {
        if ("order".equals(tableName)) {
            return "order_" + LocalDate.now().getYear();
        }
        return tableName;
    });
    
    interceptor.addInnerInterceptor(dynamicInterceptor);
    return interceptor;
}
```

### 五、多租户插件（TenantLineInnerInterceptor）
```java
@Bean
public MybatisPlusInterceptor mybatisPlusInterceptor() {
    MybatisPlusInterceptor interceptor = new MybatisPlusInterceptor();
    interceptor.addInnerInterceptor(new TenantLineInnerInterceptor(new TenantLineHandler() {
        @Override
        public Expression getTenantId() {
            return new LongValue(1L); // 从上下文获取租户ID
        }
        @Override
        public String getTenantIdColumn() {
            return "tenant_id";
        }
    }));
    return interceptor;
}

// 实体类需包含tenant_id字段
@Data
public class User {
    private Long tenantId;
}
```

### 六、数据权限插件（自定义实现）
```java
public class DataPermissionInterceptor implements InnerInterceptor {
    @Override
    public void beforeQuery(Executor executor, MappedStatement ms, 
            Object parameter, RowBounds rowBounds, ResultHandler resultHandler, BoundSql boundSql) {
        // 获取当前用户权限
        String deptCondition = " AND dept_id IN (1,2,3)";
        String newSql = boundSql.getSql() + deptCondition;
        resetSql(boundSql, newSql); // 反射修改SQL
    }
}

// 注册插件
interceptor.addInnerInterceptor(new DataPermissionInterceptor());
```

### 插件执行顺序建议
```java
interceptor.addInnerInterceptor(new TenantLineInnerInterceptor()); // 多租户第一
interceptor.addInnerInterceptor(new DynamicTableNameInnerInterceptor()); // 动态表名第二
interceptor.addInnerInterceptor(new PaginationInnerInterceptor()); // 分页第三
interceptor.addInnerInterceptor(new OptimisticLockerInnerInterceptor()); // 乐观锁第四
interceptor.addInnerInterceptor(new BlockAttackInnerInterceptor()); // 防全表最后
```

```
private void resetSql(BoundSql boundSql, String newSql) {
        try {
            // 反射修改 BoundSql 的 sql 字段
            Field sqlField = BoundSql.class.getDeclaredField("sql");
            sqlField.setAccessible(true);
            sqlField.set(boundSql, newSql);
        } catch (NoSuchFieldException | IllegalAccessException e) {
            throw new RuntimeException("反射修改 SQL 失败", e);
        }
    }
```
