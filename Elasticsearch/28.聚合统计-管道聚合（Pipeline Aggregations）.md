```

{
  "aggs": {
    "group_tag": {
      "terms": {
        "field": "tag.keyword"
      },
      "aggs": {
        "age_sum":{
          "sum": {
            "field": "age"
          }
        }
      }
    },
    "age_avg":{
      "avg_bucket": {
        "buckets_path": "group_tag>age_sum"
      }
    }
  }
}
```
