## 普通对象数组搜索存在的问题
准备一点数据
```
PUT example_demo_array
{
  "mappings": {
    "properties": {
      "order_no":{"type": "keyword"},
      "order_time":{"type": "date","format": ["yyyy-MM-dd HH:mm:ss"]},
      "good_info":{
        "type": "object", 
        "properties": {
          "good_name":{"type":"text","fields": {"keyword":{"type":"keyword"}}},
          "price":{"type":"float"},
          "count":{"type":"integer"}
        }
      }
    }
  }
}
```
```
POST example_demo_array/_bulk
{"index":{}}
{"order_no":"10001","order_time":"2023-08-03 10:20:02","good_info":[{"good_name":"联想笔记本","price":30.5,"count":1},{"good_name":"华硕笔记本","price":10.5,"count":2}]}
```
ok,现在我们想搜索一条 条件永远不满足的数据，看是否能搜索出来
good_name like '联想' and count >1
```
POST example_demo_array/_search
{
  "query": {
    "bool": {
      "must": [
        {
          "match": {
            "good_info.good_name": "联想"
          }
        },
        {
          "range": {
            "good_info.count": {
              "gt": 1
            }
          }
        }
      ]
    }
  }
}
```

## Nested 使用
准备数据
```
PUT example_demo_nested
{
  "mappings": {
    "properties": {
      "order_no":{"type": "keyword"},
      "order_time":{"type": "date","format": ["yyyy-MM-dd HH:mm:ss"]},
      "good_info":{
        "type": "nested", 
        "properties": {
          "good_name":{"type":"text","fields": {"keyword":{"type":"keyword"}}},
          "price":{"type":"float"},
          "count":{"type":"integer"}
        }
      }
    }
  }
}
```
```
POST example_demo_nested/_bulk
{"index":{}}
{"order_no":"10001","order_time":"2023-08-03 10:20:02","good_info":[{"good_name":"联想笔记本","price":30.5,"count":1},{"good_name":"华硕笔记本","price":10.5,"count":2}]}
{"index":{}}
{"order_no":"10002","order_time":"2023-08-04 10:20:02","good_info":[{"good_name":"苹果笔记本","price":230.5,"count":1},{"good_name":"苹果手机","price":60.5,"count":2}]}
```
### Nested 搜索
```
POST example_demo_nested/_search
{
  "query": {
    "nested": {
      "inner_hits": {
        "highlight": {
          "fields": {"*":{}}
        }
      }, 
      "path": "good_info",
      "query": {
        "bool": {
      "must": [
        {
          "match": {
            "good_info.good_name": "联想"
          }
        },
        {
          "range": {
            "good_info.count": {
              "gte": 1
            }
          }
        }
      ]
    }
      }
    }
  }
}
```
### 按照嵌套对象字段排序
因为嵌套对以数组形式存在父文档中，所以如果要用嵌套对象字段排序，必须算出度量值，如 最大值(max)、最小值(min)、求和(sum)、平均值(avg)
```
POST example_demo_nested/_search
{ 
  "query": {
  "match_all": {}
  },
  "sort": [
    {
      "good_info.price": {
        "order": "desc",
        "nested": {"path":"good_info"},
        "mode": "sum"
      }
    }
  ]
}
```

### 嵌套聚合
查询所有订单中，每个商品被买的数量
```
POST example_demo_nested/_search
{
  "aggs": {
    "aggs_good_name": {
      "nested": {
        "path": "good_info"
      }, 
      "aggs": {
        "nested_good_name": {
          "terms": {
            "field": "good_info.price"
          }
        }
      }
    }
  }
}
```
### 反转嵌套聚合
基于上面的基础上，查询每个商品都有哪些人买，并且统计每个购买这款商品的数量
```
POST example_demo_nested/_search
{
  "aggs": {
    "aggs_good_name": {
      "nested": {
        "path": "good_info"
      },
      "aggs": {
        "nested_good_name": {
          "terms": {
            "field": "good_info.good_name.keyword"
          },
          "aggs": {
            "reverse": {
              "reverse_nested": {},
              "aggs": {
                "agg_buyer": {
                  "terms": {
                    "field": "buyer.keyword"
                  }
                }
              }
            }
          }
        }
      }
    }
  }
}
```
